1. 사용자용 (배출업체 담당자) — ./waste.sh
---------------------------------------------------------------------
./waste.sh new 입력서.docx                      # ① 입력서 접수 → 폐기물 목록 (1번을 작업 중으로)
./waste.sh new 입력서.docx --synthetic          #    가상 입력서로 연습할 때만 --synthetic
./waste.sh cases                                # 폐기물 목록 (▶ = 작업 중)
./waste.sh use 2                                # 작업할 폐기물 바꾸기 (번호 또는 case_id)

./waste.sh next                                 # ② 파트 1이 더 묻는 질문 (한 번에 최대 3개, 번호 붙음)
./waste.sh answer 1 "2026-11-09"                # 질문 번호로 답하기 (판 번호 자동)
./waste.sh next                                 # "더 물을 질문이 없습니다" 나올 때까지
./waste.sh code 51-03-01 "시행규칙 별표 4 폐합성수지류"   # 분류번호를 근거와 함께 기록

./waste.sh send                                 # ③ 파트 2 판단 → 결과 바로 표시
./waste.sh send -v                              #    진행 과정을 쉬운 말로 보면서
./waste.sh send --queue                         #    감시(start)를 켜 둔 경우: inbox 에만 넣기
./waste.sh result                               # ④ 결과 다시 보기
                                                #    경로 · 가공업체 순위 · 운송업체 순위 · 필요한 서류 · 더 알려줄 것

  연습 순서 (가상 입력서):
  ./waste.sh new src/part1/fixtures/intake_records_example.docx --synthetic
  ./waste.sh next
  ./waste.sh answer 1 "2026-11-09"
  ./waste.sh next
  ./waste.sh code 51-03-01 "시행규칙 별표 4 폐합성수지류"
  ./waste.sh send -v

2. 관리자용 — ./waste-admin.sh
---------------------------------------------------------------------
./waste-admin.sh start -v                       # 파트 2 감시 켜기 (inbox → outbox), 쉬운 말 로그
./waste-admin.sh start -vv                      #   노드·시간·호출 수 로그
./waste-admin.sh log                            # 실시간 로그 따라 보기 (다른 창에서)
./waste-admin.sh status                         # 대기·완료·실패·결과 목록, 작업 중 사례
./waste-admin.sh web                            # 관리자 웹 화면 → http://127.0.0.1:8765/admin
./waste-admin.sh detail                         # 작업 중 사례의 판단 전체 (터미널)
./waste-admin.sh detail --case 사례ID           # 다른 사례
./waste-admin.sh detail --case 사례ID --q B-1   # 질문 하나 자세히 (근거·모델별 이유·인용)
./waste-admin.sh questions                      # 질문 번호표 (A-1 … J-4 가 무슨 질문인지)
./waste-admin.sh p1 part1.py status             # 파트 1 명령 그대로 (--db·--case 자동)
./waste-admin.sh models                         # 쓸 수 있는 모델 · 설정 모델 OK 확인
./waste-admin.sh eval                           # 사전검증 24건 평가 (모델 비용 있음)
./waste-admin.sh cache                          # dry(가짜) 답 캐시 지우기
./waste-admin.sh test                           # 테스트

  관리자만 테스트 데이터로 확인 (창 2개):
